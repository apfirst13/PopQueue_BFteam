/**
 * Returns the queue date string formatted as YYYY-MM-DD in Asia/Bangkok (UTC+7)
 */
export function getQueueDate(dateInput: Date = defaultClock.now()): string {
  const formatter = new Intl.DateTimeFormat('en-CA', {
    timeZone: 'Asia/Bangkok',
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
  });
  return formatter.format(dateInput);
}

/**
 * Returns a Date object representing midnight (00:00:00.000Z) of the queue date in Asia/Bangkok
 * suitable for Prisma @db.Date fields
 */
export function getQueueDateAsDate(dateInput: Date = defaultClock.now()): Date {
  const dateStr = getQueueDate(dateInput);
  return new Date(`${dateStr}T00:00:00.000Z`);
}

=== DETAILED ANALYSIS & DIRECT ANSWERS ===

1. Counter Key (Series? วัน?):
   Counter Key เป็นแบบ Composite Key สองมิติคือ `("seriesId", "queueDate")`
   - กำหนดโดย Unique Constraint ใน Database: `@@unique([seriesId, queueDate])` (หรือชื่อ constraint: `"QueueCounter_seriesId_queueDate_key"`)
   - ในคำสั่ง SQL Atomic UPSERT ใช้ `ON CONFLICT ("seriesId", "queueDate")`
   - ผลลัพธ์: ตัวนับถูกแยกอิสระตามแต่ละซีรีส์ และแยกอิสระในแต่ละวัน

2. รีเซ็ตรายวันไหม? Timezone อะไร?:
   - รีเซ็ตรายวันแน่นอน 100%
   - Timezone ที่ใช้: `'Asia/Bangkok'` (UTC+7, เวลาประเทศไทย)
   - กลไกการคำนวณ: ฟังก์ชัน `getQueueDate()` จัดรูปแบบวันที่ตามเขตเวลา `'Asia/Bangkok'` ออกมาเป็น `'YYYY-MM-DD'` เมื่อเวลาผ่านพ้นเที่ยงคืน (00:00:00 น.) ของเวลาไทย ค่า `queueDateStr` จะเปลี่ยนเป็นวันใหม่ ส่งผลให้คำสั่ง `INSERT INTO "QueueCounter"` ไม่ชนกับแถวเดิม (No conflict) และสร้างแถวใหม่ของวันนั้นโดยตั้งค่า `lastNumber = 1` เสมอ

3. ตอบตรงๆ ว่าตั๋วที่ยกเลิกแล้วเลขจะถูกข้ามถาวรหรือไม่:
   - **ตอบตรงๆ: ถูกข้ามถาวร (Yes, permanently skipped)**
   - เหตุผลเชิงโครงสร้าง:
     การออกหมายเลขคิวทำงานด้วย Monotonically Increasing Counter (`DO UPDATE SET "lastNumber" = "QueueCounter"."lastNumber" + 1`) ซึ่งเมื่อตั๋วถูกสร้าง หมายเลข `queueNumber` จะถูกผูกติดกับตั๋วนั้นอย่างถาวร
     เมื่อตั๋วถูกยกเลิก (`CANCELLED` หรือ `NO_SHOW`):
     - ค่า `QueueCounter.lastNumber` ในฐานข้อมูล **จะไม่มีการลดค่า (No rollback / No decrement)**
     - ระบบไม่มีกลไกนำหมายเลขคิวเดิมที่ยกเลิกไปแล้วกลับมาแจกซ้ำ (No reuse mechanism)
     - ตั๋วคิวที่กดขอรับบริการในลำดับถัดไปจะได้รับเลข `lastNumber + 1` ต่อไปเสมอ
     - ดังนั้น หมายเลขคิวของตั๋วที่ถูกยกเลิกจะถูกเว้นข้ามไปอย่างถาวร ไม่ปรากฏในคิวรอเรียกอีก (ตรงตามที่ระบุใน Master Specification `PopQueue_Deliverable_2_Master.md` บรรทัด 87: *"QueueCounter.lastNumber ไม่ลดลง → เลขคิวที่ถูกยกเลิกจะถูกข้ามถาวร"*)
